## 添加新工具

### 工具定义格式

在 `src/tools/default-tools.js` 中添加：

```javascript
{
  // 1. 工具名称（唯一标识）
  name: 'browser_custom_action',
  
  // 2. 工具描述（LLM 会看到这个描述）
  description: '自定义浏览器操作，用于特殊场景',
  
  // 3. 参数定义（JSON Schema 格式）
  schema: {
    target: {
      type: 'string',
      description: '目标元素选择器',
      required: true
    },
    action: {
      type: 'string',
      description: '执行的操作',
      enum: ['click', 'hover', 'focus'],
      default: 'click'
    },
    options: {
      type: 'object',
      description: '额外选项',
      required: false
    }
  },
  
  // 4. 处理函数（实际执行逻辑）
  handler: async (args, context) => {
    const { target, action = 'click', options = {} } = args;
    const { browser } = context;
    
    try {
      // 执行具体操作
      if (action === 'click') {
        await browser.click(target, options);
      } else if (action === 'hover') {
        await browser.hover(target);
      } else if (action === 'focus') {
        await browser.evaluate(sel => {
          document.querySelector(sel).focus();
        }, target);
      }
      
      // 返回成功结果
      return {
        success: true,
        message: `成功执行 ${action} 操作于 ${target}`,
        timestamp: Date.now()
      };
      
    } catch (error) {
      // 返回错误结果
      return {
        success: false,
        error: error.message,
        suggestion: '尝试使用其他选择器或等待元素加载'
      };
    }
  }
}
```

  } else if (this.provider === 'gemini') {

async _callOllama(messages, options) {
  const response = await fetch(`${this.client.endpoint}/api/chat`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      model: this.model,
      messages: messages,
      stream: false
    })
  });
  
  const data = await response.json();
  return {
    type: 'text',
    content: data.message.content
  };
}

### WebSocket 实时通信

编辑 `server/websocket-server.js`:

```javascript
const WebSocket = require('ws');

class WebSocketServer {
  constructor(port) {
    this.wss = new WebSocket.Server({ port });
    this.clients = new Set();
    this.setupHandlers();
  }
  
  setupHandlers() {
    this.wss.on('connection', (ws) => {
      this.clients.add(ws);
      console.log('✅ 客户端已连接');
      
      ws.on('message', async (message) => {
        const data = JSON.parse(message);
        
        if (data.type === 'execute_goal') {
          // 执行目标并推送进度
          await this.executeWithProgress(ws, data.goal);
        }
      });
      
      ws.on('close', () => {
        this.clients.delete(ws);
        console.log('客户端已断开');
      });
    });
  }
  
  async executeWithProgress(ws, goal) {
    // 推送开始消息
    ws.send(JSON.stringify({
      type: 'progress',
      status: 'started',
      goal: goal
    }));
    
    // 执行并推送进度
    agent.on('step_completed', (step) => {
      ws.send(JSON.stringify({
        type: 'progress',
        status: 'in_progress',
        step: step,
        percent: step.index / step.total * 100
      }));
    });
    
    // 推送完成消息
    const result = await agent.executeGoal(goal);
    ws.send(JSON.stringify({
      type: 'completed',
      result: result
    }));
  }
  
  broadcast(message) {
    this.clients.forEach(client => {
      if (client.readyState === WebSocket.OPEN) {
        client.send(message);
      }
    });
  }
}
```

## 性能优化

### 1. 并发控制

```javascript
class AgentPool {
  constructor(maxSize = 10) {
    this.maxSize = maxSize;
    this.agents = [];
    this.queue = [];
  }
  
  async acquire() {
    if (this.agents.length < this.maxSize) {
      const agent = new AutonomousAgent();
      await agent.launch();
      this.agents.push(agent);
      return agent;
    }
    
    // 等待可用 agent
    return new Promise(resolve => {
      this.queue.push(resolve);
    });
  }
  
  release(agent) {
    const index = this.agents.indexOf(agent);
    if (index > -1) {
      this.agents.splice(index, 1);
    }
    
    if (this.queue.length > 0) {
      const resolve = this.queue.shift();
      resolve(agent);
    }
  }
}
```

### 3. 内存管理

```javascript
class MemoryManager {
  constructor(maxMemory = 1024 * 1024 * 1024) {
    this.maxMemory = maxMemory;
    this.episodes = [];
  }
  
  addEpisode(episode) {
    this.episodes.push(episode);
    
    // 检查内存使用
    const memoryUsage = process.memoryUsage();
    if (memoryUsage.heapUsed > this.maxMemory) {
      this.cleanup();
    }
  }
  
  cleanup() {
    // 保留最近 100 个 episode
    const keepCount = 100;
    const removed = this.episodes.splice(0, this.episodes.length - keepCount);
    
    console.log(`清理了 ${removed.length} 个旧 episode`);
    
    // 触发 GC
    if (global.gc) {
      global.gc();
    }
  }
}
```

function saveEpisode(episode) {
